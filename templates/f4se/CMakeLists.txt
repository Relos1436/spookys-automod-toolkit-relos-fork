cmake_minimum_required(VERSION 3.24)
project(@NAME@ LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This plugin requires Microsoft C++ and a 64-bit target (-A x64).")
endif()
set(F4SE_SDK_ROOT "" CACHE PATH "SDK directory containing f4se/PluginAPI.h and f4se_common/")
find_path(F4SE_COMMON_ROOT common/ITypes.h PATHS "${F4SE_SDK_ROOT}/.." "${F4SE_SDK_ROOT}/../common" NO_DEFAULT_PATH)
if(NOT EXISTS "${F4SE_SDK_ROOT}/f4se/PluginAPI.h" OR NOT F4SE_COMMON_ROOT)
  message(FATAL_ERROR "Set F4SE_SDK_ROOT to the matching official SDK source directory.")
endif()
add_library(plugin_sdk INTERFACE)
target_include_directories(plugin_sdk SYSTEM INTERFACE "${F4SE_SDK_ROOT}" "${F4SE_COMMON_ROOT}")
target_compile_features(plugin_sdk INTERFACE cxx_std_17)
target_compile_options(plugin_sdk INTERFACE /EHsc)
target_compile_definitions(plugin_sdk INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(${PROJECT_NAME} SHARED main.cpp)
target_link_libraries(${PROJECT_NAME} PRIVATE plugin_sdk)
set_property(TARGET ${PROJECT_NAME} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(${PROJECT_NAME} PRIVATE /W4 /permissive-)
install(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION F4SE/Plugins)
include(CTest)
if(BUILD_TESTING)
  add_executable(plugin_host host.cpp)
  target_link_libraries(plugin_host PRIVATE plugin_sdk)
  add_test(NAME plugin_contract COMMAND plugin_host "$<TARGET_FILE:${PROJECT_NAME}>")
endif()
